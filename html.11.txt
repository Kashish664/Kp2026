    <h1>HTML5 APIs Showcase</h1>

    <!-- 1. Drag and Drop API -->
    <section>
        <h2>1. Drag & Drop</h2>
        <p>Grab the blue box and drop it into the dashed area.</p>
        <div id="drag-item" draggable="true">Drag Me!</div>
        <div id="drop-zone">Drop Here</div>
    </section>

    <!-- 2. Geolocation API -->
    <section>
        <h2>2. Geolocation</h2>
        <p>Click to find your current coordinates.</p>
        <button onclick="getLocation()">Get My Location</button>
        <p id="geo-output">Location details will appear here...</p>
    </section>

    <!-- 3. Web Storage API (Local Storage) -->
    <section>
        <h2>3. Web Storage</h2>
        <p>Data saved here persists even after refreshing the page.</p>
        <input type="text" id="name-input" placeholder="Enter your name">
        <button onclick="saveName()">Save to LocalStorage</button>
        <button onclick="clearName()" style="background:#e74c3c">Clear Storage</button>
        <p id="storage-output"></p>
    </section>

    <script>
        /** 
         * DRAG AND DROP API
         * Uses 'draggable' attribute and events like 'dragstart', 'dragover', and 'drop'.
         */
        const dragItem = document.getElementById('drag-item');
        const dropZone = document.getElementById('drop-zone');

        dragItem.addEventListener('dragstart', (e) => {
            e.dataTransfer.setData('text/plain', 'Box successfully dropped!');
        });

        dropZone.addEventListener('dragover', (e) => {
            e.preventDefault(); // Required to allow a drop
            dropZone.classList.add('hover');
        });

        dropZone.addEventListener('dragleave', () => dropZone.classList.remove('hover'));

        dropZone.addEventListener('drop', (e) => {
            e.preventDefault();
            const data = e.dataTransfer.getData('text/plain');
            dropZone.textContent = data;
            dropZone.classList.remove('hover');
            dragItem.style.opacity = '0.5';
        });

        /** 
         * GEOLOCATION API
         * Uses 'navigator.geolocation' to request user coordinates.
         */
        function getLocation() {
            const output = document.getElementById('geo-output');
            if (navigator.geolocation) {
                output.textContent = "Requesting permission...";
                navigator.geolocation.getCurrentPosition(
                    (pos) => {
                        output.innerHTML = `<strong>Latitude:</strong> ${pos.coords.latitude}<br><strong>Longitude:</strong> ${pos.coords.longitude}`;
                    },
                    (err) => { output.textContent = `Error: ${err.message}`; }
                );
            } else {
                output.textContent = "Geolocation is not supported by this browser.";
            }
        }

        /** 
         * WEB STORAGE API
         * Uses 'localStorage' to store key-value pairs persistently.
         */
        const nameInput = document.getElementById('name-input');
        const storageOutput = document.getElementById('storage-output');

        // Check for existing data on load
        window.onload = () => {
            const savedName = localStorage.getItem('demoName');
            if (savedName) storageOutput.textContent = `Welcome back, ${savedName}!`;
        };

        function saveName() {
            const name = nameInput.value;
            if (name) {
                localStorage.setItem('demoName', name);
                storageOutput.textContent = `Name "${name}" saved! Refresh the page to see.`;
            }
        }

        function clearName() {
            localStorage.removeItem('demoName');
            storageOutput.textContent = "Storage cleared.";
            nameInput.value = "";
        }
    </script>
